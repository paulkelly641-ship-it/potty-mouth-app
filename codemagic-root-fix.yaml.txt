workflows:
  ios-workflow:
    name: Potty Mouth iOS Release
    instance_type: mac_mini_m2
    max_build_duration: 90
    environment:
      groups:
        - appstore
      vars:
        BUNDLE_ID: "com.paulkelly.pottymouthsongwriter"
        XCODE_PROJECT: "ios/App/App.xcodeproj"
        XCODE_SCHEME: "App"
      node: 22.12.0
      xcode: 16.2
      cocoapods: default
    scripts:
      - name: Install npm dependencies
        script: |
          npm ci || npm install
      - name: Build web assets
        script: |
          if [ -f "index.html" ] && [ ! -d "www" ]; then mkdir -p www; cp index.html www/; fi
          npm run build || true
      - name: FIX Bundle ID in ALL files BEFORE iOS add
        script: |
          echo "=== Fixing capacitor.config.json ==="
          if [ -f "capacitor.config.json" ]; then
            cat capacitor.config.json
            sed -i '' "s/com.yourname.pottymouth/$BUNDLE_ID/g" capacitor.config.json 2>/dev/null || sed -i "s/com.yourname.pottymouth/$BUNDLE_ID/g" capacitor.config.json
            sed -i '' "s/io.ionic.starter/$BUNDLE_ID/g" capacitor.config.json 2>/dev/null || sed -i "s/io.ionic.starter/$BUNDLE_ID/g" capacitor.config.json
            cat capacitor.config.json
          fi
          if [ -f "capacitor.config.ts" ]; then
            sed -i '' "s/com.yourname.pottymouth/$BUNDLE_ID/g" capacitor.config.ts 2>/dev/null || sed -i "s/com.yourname.pottymouth/$BUNDLE_ID/g" capacitor.config.ts
            sed -i '' "s/io.ionic.starter/$BUNDLE_ID/g" capacitor.config.ts 2>/dev/null || sed -i "s/io.ionic.starter/$BUNDLE_ID/g" capacitor.config.ts
          fi
          if [ -f "package.json" ]; then
            sed -i '' "s/com.yourname.pottymouth/$BUNDLE_ID/g" package.json 2>/dev/null || true
          fi
      - name: Add iOS platform
        script: |
          rm -rf ios
          npm install @capacitor/core @capacitor/cli @capacitor/ios
          npx cap add ios
          npx cap sync ios
          echo "=== After sync, fix Xcode proj again ==="
          sed -i '' "s/com.yourname.pottymouth/$BUNDLE_ID/g" ios/App/App.xcodeproj/project.pbxproj 2>/dev/null || sed -i "s/com.yourname.pottymouth/$BUNDLE_ID/g" ios/App/App.xcodeproj/project.pbxproj
          sed -i '' "s/io.ionic.starter/$BUNDLE_ID/g" ios/App/App.xcodeproj/project.pbxproj 2>/dev/null || sed -i "s/io.ionic.starter/$BUNDLE_ID/g" ios/App/App.xcodeproj/project.pbxproj
          /usr/libexec/PlistBuddy -c "Set :CFBundleIdentifier $BUNDLE_ID" "ios/App/App/Info.plist" 2>/dev/null || true
          echo "=== VERIFY BUNDLE ID ==="
          grep -r "PRODUCT_BUNDLE_IDENTIFIER" ios/App/App.xcodeproj/ | head -5
      - name: Set encryption flag
        script: |
          /usr/libexec/PlistBuddy -c "Add :ITSAppUsesNonExemptEncryption bool false" "ios/App/App/Info.plist" 2>/dev/null || /usr/libexec/PlistBuddy -c "Set :ITSAppUsesNonExemptEncryption false" "ios/App/App/Info.plist"
      - name: Save API key
        script: |
          mkdir -p /tmp
          python3 -c "
          import os
          data = os.environ.get('APP_STORE_CONNECT_PRIVATE_KEY','')
          data = data.replace('\\n','\n')
          open('/tmp/AuthKey.p8','w').write(data.strip()+'\n')
          print('Key fixed to /tmp/AuthKey.p8')
          "
          ls -lh /tmp/AuthKey.p8
      - name: Set up code signing
        script: |
          set -x
          keychain initialize
          app-store-connect create-bundle-id $BUNDLE_ID "Potty Mouth" --key-id $APP_STORE_CONNECT_KEY_IDENTIFIER --issuer-id $APP_STORE_CONNECT_ISSUER_ID --private-key @/tmp/AuthKey.p8 || echo "bundle exists"
          app-store-connect fetch-signing-files $BUNDLE_ID --type IOS_APP_STORE --create --key-id $APP_STORE_CONNECT_KEY_IDENTIFIER --issuer-id $APP_STORE_CONNECT_ISSUER_ID --private-key @/tmp/AuthKey.p8
          keychain add-certificates
          echo "=== PROFILES ==="
          ls -lh ~/Library/MobileDevice/Provisioning\ Profiles/ | head -20
      - name: Build IPA
        script: |
          set -x
          xcode-project use-profiles
          xcode-project build-ipa --project "$XCODE_PROJECT" --scheme "$XCODE_SCHEME"
          ls -lh build/ios/ipa/
    artifacts:
      - build/ios/ipa/*.ipa
      - /tmp/xcodebuild_logs/*.log
    publishing:
      app_store_connect:
        api_key: $APP_STORE_CONNECT_PRIVATE_KEY
        key_id: $APP_STORE_CONNECT_KEY_IDENTIFIER
        issuer_id: $APP_STORE_CONNECT_ISSUER_ID
        submit_to_testflight: true
        submit_to_app_store: false
